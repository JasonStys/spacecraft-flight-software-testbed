# Software-only executive by Jason Stys. Targets/options and variables: docs/CODE_INDEX.md.
cmake_minimum_required(VERSION 3.28)
project(SpacecraftFlightTestbed VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(FLIGHT_SANITIZE "Enable Clang ASan and UBSan" OFF)
option(FLIGHT_COVERAGE "Enable GCC coverage" OFF)
if(FLIGHT_SANITIZE AND MSVC)
  message(FATAL_ERROR "Combined sanitizer mode requires the documented Linux Clang toolchain")
endif()
if(FLIGHT_COVERAGE AND NOT CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
  message(FATAL_ERROR "Coverage mode requires GCC; refusing an uninstrumented build")
endif()
add_library(flight_model native/core.cpp native/checkpoint.cpp)
target_include_directories(flight_model PUBLIC native)
add_executable(flight_core native/cli.cpp)
target_link_libraries(flight_core PRIVATE flight_model)
add_executable(core_tests native/tests.cpp)
target_link_libraries(core_tests PRIVATE flight_model)
foreach(target flight_model flight_core core_tests)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(FLIGHT_SANITIZE)
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
  if(FLIGHT_COVERAGE)
    target_compile_options(${target} PRIVATE --coverage -O0 -g)
    target_link_options(${target} PRIVATE --coverage)
  endif()
endforeach()
enable_testing()
add_test(NAME executive_contracts COMMAND core_tests)
add_test(NAME cli_nominal COMMAND flight_core nominal 240 120 5 8 2)
add_test(NAME cli_invalid COMMAND flight_core nominal nan 120 5 8 2)
set_tests_properties(cli_invalid PROPERTIES WILL_FAIL TRUE)
